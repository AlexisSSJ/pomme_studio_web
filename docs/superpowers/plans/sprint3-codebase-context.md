### 1.3 Sync vs async SQLAlchemy

**Fully async.** `/workspaces/pomme_studio_web/backend/app/db/session.py` (verbatim):

```python
"""Async SQLAlchemy engine and request-scoped session dependency."""

from collections.abc import AsyncGenerator

from sqlalchemy.ext.asyncio import (
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)

from ..core.config import settings

# The engine is created lazily by SQLAlchemy: no connection is opened until the
# first query, so importing this module is safe without a running database.
engine = create_async_engine(
    settings.DATABASE_URL,
    echo=False,
    pool_pre_ping=True,
)

AsyncSessionLocal = async_sessionmaker(
    bind=engine,
    class_=AsyncSession,
    expire_on_commit=False,
    autoflush=False,
)


async def get_db() -> AsyncGenerator[AsyncSession, None]:
    """Yield a database session and roll back on unexpected errors."""
    async with AsyncSessionLocal() as session:
        try:
            yield session
        except Exception:
            await session.rollback()
            raise
```

- `AsyncSession`, `async_sessionmaker`, `create_async_engine`; `expire_on_commit=False`, `autoflush=False`.
- All endpoints use `db: AsyncSession = Depends(get_db)` and `await db.execute(select(...))`, `await db.get(...)`, `await db.commit()`, `await db.refresh(...)`.
- `Base` is a plain `DeclarativeBase` in `/workspaces/pomme_studio_web/backend/app/db/base.py` (verbatim):

## 2. `/workspaces/pomme_studio_web/backend/app/api/deps.py` — VERBATIM FULL CONTENT

```python
"""Shared FastAPI dependencies: database session and RBAC guards.

Every protected endpoint must authenticate the caller, confirm `is_active`,
and (where relevant) check the role and project membership/ownership. The
helpers here cover the first two; project-level checks will build on them.
"""

from uuid import UUID

from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from ..core.config import API_V1_PREFIX, settings
from ..core.security import decode_access_token
from ..db.session import get_db
from ..models.user import User, UserRole
from ..models.project_member import ProjectMember
from ..services.google_calendar import GoogleCalendarClient

oauth2_scheme = OAuth2PasswordBearer(
    tokenUrl=f"{API_V1_PREFIX}/auth/login",
    auto_error=False,
)


def _credentials_exception() -> HTTPException:
    return HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )


async def get_current_user(
    token: str | None = Depends(oauth2_scheme),
    db: AsyncSession = Depends(get_db),
) -> User:
    """Resolve the authenticated user from the bearer token."""
    if not token:
        raise _credentials_exception()

    payload = decode_access_token(token)
    if payload is None:
        raise _credentials_exception()

    subject = payload.get("sub")
    try:
        user_id = UUID(str(subject))
    except (TypeError, ValueError):
        raise _credentials_exception() from None

    user = await db.get(User, user_id)
    if user is None:
        raise _credentials_exception()
    return user


async def get_current_active_user(
    current_user: User = Depends(get_current_user),
) -> User:
    """Reject users that have been deactivated (kill switch)."""
    if not current_user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Inactive user",
        )
    return current_user


def require_roles(*roles: UserRole):
    """Dependency factory enforcing that the caller holds one of `roles`."""

    async def _checker(
        current_user: User = Depends(get_current_active_user),
    ) -> User:
        if current_user.role not in roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Insufficient permissions",
            )
        return current_user

    return _checker


def require_project_access(project_id: UUID):
    """Dependency factory enforcing project membership or SUPER_ADMIN access.

    Usage with a fixed project_id:
        Depends(require_project_access(some_uuid))

    For path-parameter-based access, use `require_project_access_from_path`.
    """

    async def _checker(
        current_user: User = Depends(get_current_active_user),
        db: AsyncSession = Depends(get_db),
    ) -> User:
        if current_user.role == UserRole.SUPER_ADMIN:
            return current_user

        result = await db.execute(
            select(ProjectMember).where(
                ProjectMember.project_id == project_id,
                ProjectMember.user_id == current_user.id,
            )
        )
        member = result.scalar_one_or_none()
        if member is None:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="No access to this project",
            )
        return current_user

    return _checker


async def require_project_access_from_path(
    project_id: UUID,
    current_user: User = Depends(get_current_active_user),
    db: AsyncSession = Depends(get_db),
) -> User:
    """Dependency that reads project_id from the path and checks access."""
    if current_user.role == UserRole.SUPER_ADMIN:
        return current_user

    result = await db.execute(
        select(ProjectMember).where(
            ProjectMember.project_id == project_id,
            ProjectMember.user_id == current_user.id,
        )
    )
    member = result.scalar_one_or_none()
    if member is None:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="No access to this project",
        )
    return current_user


def get_calendar_client() -> GoogleCalendarClient:
    """Return the process-wide Google Calendar client instance."""
    return GoogleCalendarClient(settings)
```

**Key signatures to reuse in Sprint 3:**
- `oauth2_scheme = OAuth2PasswordBearer(tokenUrl=f"{API_V1_PREFIX}/auth/login", auto_error=False)`
- `_credentials_exception() -> HTTPException` → 401 `{"detail": "Could not validate credentials"}`, header `WWW-Authenticate: Bearer`
- `async def get_current_user(token: str | None = Depends(oauth2_scheme), db: AsyncSession = Depends(get_db)) -> User`
- `async def get_current_active_user(current_user: User = Depends(get_current_user)) -> User` → 403 `"Inactive user"`
- `def require_roles(*roles: UserRole)` → returns async `_checker(current_user=Depends(get_current_active_user)) -> User` → 403 `"Insufficient permissions"`
- `def require_project_access(project_id: UUID)` → returns async `_checker(current_user, db) -> User`; SUPER_ADMIN bypass; else requires a `ProjectMember` row → 403 `"No access to this project"`
- `async def require_project_access_from_path(project_id: UUID, current_user=Depends(get_current_active_user), db=Depends(get_db)) -> User` — same logic, path-param friendly
- `def get_calendar_client() -> GoogleCalendarClient` — returns `GoogleCalendarClient(settings)` (tests override via `app.dependency_overrides`)

**Important:** `get_db` is imported into `deps.py` from `..db.session` and re-exported there; endpoints import `get_db` from either `....api.deps` (leads.py) or `....db.session` (admin_projects.py). Both work.

### 3.6 Patterns summary (apply these in Sprint 3)

- **Enum pattern:** `class X(str, enum.Enum)` with UPPERCASE members and lowercase string values for domain enums (`ProjectStatus`, `LeadStatus`, `ServiceType`, `SpaceType`, `BudgetRange`, `MeetingType`, `MeetingStatus`); UPPERCASE string values for RBAC enums (`UserRole`, `ProjectRole`). Persisted with `SAEnum(EnumClass, name="<snake_case_name>", native_enum=False)`. **Exception:** `Meeting.status` adds `values_callable=lambda members: [member.value for member in members]` so lowercase values are stored (see §3.7).
- **UUID PK pattern:** `id: Mapped[uuid.UUID] = mapped_column(Uuid(as_uuid=True), primary_key=True, default=uuid.uuid4)`.
- **Timestamp pattern:** `created_at` = `DateTime(timezone=True), default=lambda: datetime.now(timezone.utc), nullable=False`; `updated_at` adds `onupdate=lambda: datetime.now(timezone.utc)`.
- **Base import path:** `from ..db.base import Base` (from a model file inside `backend/app/models/`).
- **No SQLAlchemy relationships** are defined anywhere — all linkage is manual via FK columns and explicit `select`/`db.get`. No `relationship()`, no `cascade`, no ORM-level membership helpers.
- **No `deliverables`, `ffe_items`, or `payments` models exist** despite being listed in `docs/architecture.md` §4.3 as "Sprint 1" — those were never implemented. Only 7 tables exist: `users`, `projects`, `project_members`, `pricing_config`, `audit_logs`, `leads`, `meetings`.

```python
"""Admin project activation and internal listing endpoints."""

from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException, status

from ....schemas.project import ProjectRead

@router.get("/projects", response_model=list[ProjectRead])
async def list_projects(
    current_user: User = Depends(require_roles(UserRole.SUPER_ADMIN, UserRole.LEAD_DESIGNER)),
    db: AsyncSession = Depends(get_db),
) -> list[Project]:
    """List internal projects.

    SUPER_ADMIN sees every project; LEAD_DESIGNER sees projects where they are
    the assigned lead designer or a project member.
    """
    if current_user.role == UserRole.SUPER_ADMIN:
        result = await db.execute(
            select(Project).order_by(Project.created_at.desc())
        )
    else:
        result = await db.execute(
            select(Project)
            .where(
                (Project.lead_designer_id == current_user.id)
                | Project.id.in_(
                    select(ProjectMember.project_id).where(
                        ProjectMember.user_id == current_user.id
                    )
                )
            )
            .order_by(Project.created_at.desc())
        )
    return list(result.scalars().all())


@router.post("/projects/{project_id}/activate", response_model=ProjectRead)
async def activate_project(
    project_id: UUID,
    current_user: User = Depends(require_roles(UserRole.SUPER_ADMIN)),
    db: AsyncSession = Depends(get_db),
) -> Project:
    """Activate a DRAFT project after the down payment is received.

    Sets the project status to ACTIVE, marks the balance as paid and creates
    CLIENT / LEAD_DESIGNER memberships from the project's client_id and
    lead_designer_id when present, skipping users that already have a
    membership for the project.
    """
    project = await db.get(Project, project_id)
    if project is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Project not found",
        )

    if project.status != ProjectStatus.DRAFT:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Project is not in draft status",
        )

    project.balance_paid = True
    project.status = ProjectStatus.ACTIVE

    desired_members: list[tuple[UUID, ProjectRole]] = []
    if project.client_id is not None:
        desired_members.append((project.client_id, ProjectRole.CLIENT))
    if project.lead_designer_id is not None:
        desired_members.append((project.lead_designer_id, ProjectRole.LEAD_DESIGNER))

    members_added: list[str] = []
    if desired_members:
        desired_user_ids = [user_id for user_id, _ in desired_members]
        existing_result = await db.execute(
            select(ProjectMember.user_id).where(
                ProjectMember.project_id == project.id,
                ProjectMember.user_id.in_(desired_user_ids),
            )
        )
        existing_user_ids = set(existing_result.scalars().all())

        for user_id, role in desired_members:
            if user_id not in existing_user_ids:
                db.add(
                    ProjectMember(
                        project_id=project.id,
                        user_id=user_id,
                        assigned_role=role,
                    )
                )
                members_added.append(str(user_id))

    try:
        await db.commit()
    except IntegrityError as exc:
        await db.rollback()
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Project is not in draft status",
        ) from exc
    await db.refresh(project)

    await log_audit(
        db,
        actor_user_id=current_user.id,
        action="project.activate",
        entity_type="project",
        entity_id=project.id,
        metadata={
            "balance_paid": True,
            "members_added": members_added,
        },
    )

    return project
```

### 5.4 Error-handling / response / audit patterns (summary)

- **Error handling:** explicit `HTTPException(status_code=status.HTTP_XXX, detail="...")`; `raise ... from exc` when wrapping domain exceptions. ### 9.2 API calls — `/workspaces/pomme_studio_web/frontend/lib/api.ts` VERBATIM

```typescript
import { getToken, clearToken } from "./auth";

const API_BASE = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000";

export function extractDetail(data: unknown, fallback: string): string {
  if (typeof data === "string") return data;

  if (Array.isArray(data)) {
    const messages = data
      .map((item) => {
        if (
          item !== null &&
          typeof item === "object" &&
          "msg" in item &&
          typeof item.msg === "string"
        ) {
          return item.msg;
        }
        return null;
      })
      .filter((message): message is string => message !== null);
    return messages.length > 0 ? messages.join("; ") : fallback;
  }

  if (data !== null && typeof data === "object") {
    if ("msg" in data && typeof data.msg === "string") return data.msg;
    if ("detail" in data) return extractDetail(data.detail, fallback);
  }

  return fallback;
}

export async function apiFetch(
  path: string,
  options: RequestInit = {}
): Promise<Response> {
  const token = getToken();
  const headers: Record<string, string> = {
    "Content-Type": "application/json",
    ...(options.headers as Record<string, string>),
  };
  if (token) {
    headers["Authorization"] = `Bearer ${token}`;
  }

  const response = await fetch(`${API_BASE}${path}`, {
    ...options,
    headers,
  });

  if (response.status === 401) {
    clearToken();
    if (typeof window !== "undefined") {
      window.location.href = "/login";
    }
  }

  return response;
}
```

**Important for uploads:** `apiFetch` hardcodes `"Content-Type": "application/json"`. To send `FormData` for a file upload, callers must override headers (e.g. pass `headers: { "Content-Type": undefined }` or a custom wrapper) — the default is set first then `options.headers` spread, so an explicit override is required. No upload helper exists.

```tsx
"use client";

import { useEffect, useState, useCallback } from "react";
import { apiFetch, extractDetail } from "@/lib/api";

interface Project {
  id: string;
  title: string;
  lead_id: string | null;
  client_id: string | null;
  lead_designer_id: string | null;
  scope_type: string | null;
  status: string;
  included_rounds: number;
  remaining_rounds: number;
  balance_paid: boolean;
  created_at: string;
}

interface Me {
  id: string;
  email: string;
  full_name: string | null;
  specialty: string | null;
  role: string;
  is_active: boolean;
  created_at: string;
  updated_at: string;
}

export default function ProjectsPage() {
  const [projects, setProjects] = useState<Project[]>([]);
  const [me, setMe] = useState<Me | null>(null);
  const [message, setMessage] = useState("");
  const [loading, setLoading] = useState(false);
  const [activatingId, setActivatingId] = useState<string | null>(null);

  const loadProjects = useCallback(async () => {
    setLoading(true);
    try {
      const res = await apiFetch("/api/v1/admin/projects");
      if (res.ok) {
        setProjects(await res.json());
      } else {
        const data: unknown = await res.json().catch(() => ({}));
        setMessage(extractDetail(data, "Error al cargar proyectos"));
      }
    } catch {
      setMessage("Error de conexión con el servidor");
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    async function loadMe() {
      try {
        const res = await apiFetch("/api/v1/auth/me");
        if (res.ok) {
          setMe(await res.json());
        }
      } catch {
        // Silent: role-gated UI falls back to hidden controls.
      }
    }

    loadMe();
    loadProjects();
  }, [loadProjects]);

  async function activateProject(projectId: string) {
    setActivatingId(projectId);
    setMessage("");
    try {
      const res = await apiFetch(`/api/v1/admin/projects/${projectId}/activate`, {
        method: "POST",
      });
      if (res.ok) {
        setMessage("Proyecto activado");
        loadProjects();
      } else {
        const data: unknown = await res.json().catch(() => ({}));
        setMessage(extractDetail(data, "Error al activar proyecto"));
      }
    } catch {
      setMessage("Error de conexión con el servidor");
    } finally {
      setActivatingId(null);
    }
  }

  return (
    <section>
      <h1 className="text-xl tracking-tightest">Proyectos</h1>

      {message && (
        <p className="mt-4 border border-border bg-surface p-3 text-sm">{message}</p>
      )}

      {loading && projects.length === 0 ? (
        <p className="mt-4 text-sm">Cargando…</p>
      ) : (
        <div className="grid-brutal mt-4">
          {projects.map((project) => (
            <div
              key={project.id}
              className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between"

              <div>
```

## 11. Project status enum, ProjectMember semantics, meeting/lead linkage

### 11.1 Project status — what counts as "active"

`ProjectStatus` values: `DRAFT="draft"`, `ACTIVE="active"`, `ON_HOLD="on_hold"`, `COMPLETED="completed"`, `CANCELLED="cancelled"`.

- **Only `ACTIVE` ("active") counts as an active project.** There is no helper like `is_active` on `Project`; code compares `project.status == ProjectStatus.DRAFT` (activation guard in `admin_projects.py`) or `project.status == "draft"` in the frontend (`me?.role === "SUPER_ADMIN" && project.status === "draft"`).
- Default on create: `default=ProjectStatus.DRAFT`, `server_default="draft"`.
- Activation flow (`POST /api/v1/admin/projects/{id}/activate`, SUPER_ADMIN only): requires DRAFT → sets `balance_paid=True`, `status=ACTIVE`, creates CLIENT/LEAD_DESIGNER memberships, logs `project.activate`.
- `ON_HOLD`, `COMPLETED`, `CANCELLED` are defined but **never used** in any endpoint or test.
- `uq_projects_lead_id` unique constraint on `projects.lead_id` prevents duplicate lead→project conversion (multiple NULLs allowed).

### 11.2 ProjectMember role semantics — how COLLABORATOR assignment is stored

- `ProjectRole` values are UPPERCASE strings: `LEAD_DESIGNER="LEAD_DESIGNER"`, `COLLABORATOR="COLLABORATOR"`, `CLIENT="CLIENT"`. Stored via `SAEnum(ProjectRole, name="project_role", native_enum=False)` in `project_members.assigned_role` (NOT NULL).
- A user's **global** role lives in `users.role` (`UserRole`, UPPERCASE strings). A user's **per-project** role lives in `project_members.assigned_role`. They are independent: a user with global `UserRole.COLLABORATOR` gets a `ProjectMember(assigned_role=ProjectRole.COLLABORATOR)` row for each project they work on.
- `test_project` fixture demonstrates the canonical COLLABORATOR assignment:
  ```python
  member = ProjectMember(
      id=uuid.uuid4(),
      project_id=project.id,
      user_id=collaborator_user.id,
      assigned_role=ProjectRole.COLLABORATOR,
  )
  ```
- `activate_project` creates `ProjectMember` rows for `client_id` with `ProjectRole.CLIENT` and `lead_designer_id` with `ProjectRole.LEAD_DESIGNER`, skipping users who already have a membership (`uq_project_member` unique on `(project_id, user_id)`).
- **Access checks ignore `assigned_role`** — `require_project_access` / `require_project_access_from_path` only check that *a* `ProjectMember` row exists for `(project_id, user_id)`; any assigned role grants access. SUPER_ADMIN bypasses entirely.
- `list_projects` for LEAD_DESIGNER filters by `Project.lead_designer_id == user.id OR Project.id IN (SELECT project_id FROM project_members WHERE user_id = user.id)` — again role-agnostic membership.
- **No endpoint exists to create/update/delete `ProjectMember` rows** — they are only created by `activate_project` and test fixtures. Sprint 3 brief/moodboard work that needs collaborator assignment will need a new endpoint or fixture pattern.

### 11.3 Meeting / lead linkage patterns

- `Meeting` has both `lead_id` (FK→`leads.id`, nullable, indexed) and `project_id` (FK→`projects.id`, nullable, indexed), with `CheckConstraint("(lead_id IS NOT NULL) <> (project_id IS NOT NULL)", name="ck_meeting_single_parent")` — **exactly one parent, enforced at DB level**.
- Discovery calls: `book_discovery` creates `Meeting(lead_id=lead.id, project_id=None, meeting_type=MeetingType.DISCOVERY, status=MeetingStatus.SCHEDULED)`, then after Google event creation sets `google_event_id`, `meet_link`, `status=CONFIRMED`. Times persisted in UTC. Audit: `meeting.create`, `actor_user_id=None`, metadata `{lead_id, google_event_id, start_time}`.
- `Project.lead_id` links a converted lead to its project (`LeadStatus.CONVERTED` set in `convert_lead`); `uq_projects_lead_id` prevents double conversion.
- `Meeting.status` uses lowercase persisted values via `values_callable`; partial unique index `ux_meetings_start_time_active` on `start_time WHERE status != 'cancelled'` prevents double-booking (both `postgresql_where` and `sqlite_where` declared).
- `MeetingCreate` schema requires timezone-aware `start_time`; `MeetingRead` exposes `meet_link` but **not** `google_event_id`.
- `docs/architecture.md` §5.6 notes meeting cancellation/rescheduling audit is future work; §5.4 notes Google Meet auto-creation via `conferenceDataVersion=1` + `hangoutsMeet`.
